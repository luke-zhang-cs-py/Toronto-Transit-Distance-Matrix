Flask>=2.3

# TTC's realtime feed is GTFS-realtime protobuf. The official bindings
# rather than a hand-rolled wire-format decoder: parsing protobuf by hand is
# real code to get wrong, and this is the one place the app trusts bytes
# from someone else. No API key -- bustime.ttc.ca is open.
gtfs-realtime-bindings>=1.0

# The timetable is Toronto wall-clock time, so "now" is read in
# America/Toronto rather than in whatever zone the machine is set to.
# zoneinfo needs a zone database: Linux and macOS ship one, Windows does not,
# and this package is it.
tzdata; sys_platform == "win32"

# Tests only; the app itself needs nothing but Flask.
pytest>=7.4

# Also tests only, and less optional than it looks. `coverage` is what
# tests/test_published_figures.py measures the published figures with, so
# without it that test errors rather than skips -- and CI installs exactly
# this file. pytest-cov is what the documented coverage command asks for.
coverage>=7.0
pytest-cov>=4.0
